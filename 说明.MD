获取日志: Get-ChildItem "$env:APPDATA\XIVLauncherCN","$env:APPDATA\XIVLauncher" -Recurse -Include *.log,*.txt -File -ErrorAction SilentlyContinue | Where-Object { $_.Length -gt 1000 } | Sort-Object LastWriteTime -Descending | Select-Object -First 8 FullName,Length,LastWriteTime | Format-List
